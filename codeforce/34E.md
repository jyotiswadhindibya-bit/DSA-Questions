```rust
use std::io::{self, Read};

#[derive(Clone, Copy)]
struct Ball {
    x: f64,
    v: f64,
    m: f64,
    id: usize,
}

fn main() {
    let mut input = String::new();
    if io::stdin().read_to_string(&mut input).is_err() {
        return;
    }

    let mut tokens = input.split_whitespace();

    let n: usize = match tokens.next() {
        Some(val) => val.parse().unwrap(),
        None => return,
    };
    let t: f64 = tokens.next().unwrap().parse().unwrap();

    let mut balls: Vec<Ball> = Vec::with_capacity(n);
    for i in 0..n {
        let x: f64 = tokens.next().unwrap().parse().unwrap();
        let v: f64 = tokens.next().unwrap().parse().unwrap();
        let m: f64 = tokens.next().unwrap().parse().unwrap();
        balls.push(Ball { x, v, m, id: i });
    }
    
    balls.sort_by(|a, b| a.x.partial_cmp(&b.x).unwrap());

    let mut current_time = 0.0;

    while current_time < t {
        let mut min_dt = 1e18;
        let mut col_idx: Option<usize> = None;

        for i in 0..(n - 1) {
            if balls[i].v > balls[i + 1].v + 1e-9 {
                let dist = (balls[i + 1].x - balls[i].x).max(0.0);
                let dt = dist / (balls[i].v - balls[i + 1].v);

                if dt < min_dt {
                    min_dt = dt;
                    col_idx = Some(i);
                }
            }
        }

        let idx = match col_idx {
            Some(i) => i,
            None => {
                let remaining_time = t - current_time;
                for ball in &mut balls {
                    ball.x += ball.v * remaining_time;
                }
                break;
            }
        };

        if current_time + min_dt >= t {
            let remaining_time = t - current_time;
            for ball in &mut balls {
                ball.x += ball.v * remaining_time;
            }
            break;
        }

        for ball in &mut balls {
            ball.x += ball.v * min_dt;
        }
        current_time += min_dt;

        let m1 = balls[idx].m;
        let m2 = balls[idx + 1].m;
        let v1 = balls[idx].v;
        let v2 = balls[idx + 1].v;

        balls[idx].v = ((m1 - m2) * v1 + 2.0 * m2 * v2) / (m1 + m2);
        balls[idx + 1].v = ((m2 - m1) * v2 + 2.0 * m1 * v1) / (m1 + m2);
    }

    balls.sort_by_key(|b| b.id);

    for ball in balls {
        println!("{:.9}", ball.x);
    }
}
```
